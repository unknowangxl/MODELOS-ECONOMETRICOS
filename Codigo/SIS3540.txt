// ---------------- INICIALIZACION ----------------

INICIO()
BEGIN
  IF SIZE(L9)<9 THEN
    L9:={0,0,0,0,0,0,0.05,4,1};
  END;
  RETURN 1;
END;


// ---------------- IDENTIFICACION DE DATOS ----------------

PIDX(t)
BEGIN
  IF t==1 THEN RETURN 1; END;
  IF t==2 THEN RETURN 2; END;
  IF t==4 THEN RETURN 3; END;
  RETURN 0;
END;


NOBS(t)
BEGIN
  LOCAL p;
  p:=PIDX(t);
  IF p==0 THEN RETURN 0; END;
  RETURN L9(2*p-1);
END;


KVAR(t)
BEGIN
  LOCAL p;
  p:=PIDX(t);
  IF p==0 THEN RETURN 0; END;
  RETURN L9(2*p);
END;


// ---------------- EDITAR MATRICES ----------------

VER(t)
BEGIN
  IF t==1 THEN EDITMAT(M1); END;
  IF t==2 THEN EDITMAT(M2); END;
  IF t==4 THEN EDITMAT(M4); END;
  RETURN 1;
END;


// ---------------- CAPTURA DE DATOS ----------------

CAPTURAR(t)
BEGIN
  LOCAL n,k,A,p;

  n:=5;
  k:=1;

  IF t==1 THEN
    IF INPUT({n},"Datos - Tema 1",
      {"n ="},
      {"Numero de observaciones"})==0 THEN
      RETURN 0;
    END;
  ELSE
    k:=2;
    IF INPUT({n,k},"Datos - Tema "+STRING(t),
      {"n =","k ="},
      {"Numero de observaciones","Numero de variables X"})==0 THEN
      RETURN 0;
    END;
  END;

  n:=IP(n);
  k:=IP(k);

  IF n<3 OR k<1 OR n<k+2 THEN
    MSGBOX("Datos insuficientes.");
    RETURN 0;
  END;

  A:=MAKEMAT(0,n,k+1);

  IF t==1 THEN M1:=A; END;
  IF t==2 THEN M2:=A; END;
  IF t==4 THEN M4:=A; END;

  p:=PIDX(t);

  L9(2*p-1):=n;
  L9(2*p):=k;

  MSGBOX("Columna 1 = Y. Columnas siguientes = X1...Xk.");

  VER(t);

  RETURN 1;
END;


// ---------------- INGRESAR O REUTILIZAR DATOS ----------------

ENTRAR(t)
BEGIN
  LOCAL op;

  op:=1;

  IF NOBS(t)==0 THEN
    RETURN CAPTURAR(t);
  END;

  IF CHOOSE(op,
    "Tema "+STRING(t),
    "Usar datos guardados",
    "Ingresar datos nuevos",
    "Volver")==0 THEN
    RETURN 0;
  END;

  IF op==1 THEN RETURN 1; END;
  IF op==2 THEN RETURN CAPTURAR(t); END;

  RETURN 0;
END;


// ---------------- SUBMENU DE DATOS ----------------

DATOS(t)
BEGIN
  LOCAL op;

  op:=1;

  IF CHOOSE(op,
    "Datos Tema "+STRING(t),
    "Ver / editar tabla",
    "Ingresar datos nuevos",
    "Atras")==0 THEN
    RETURN 0;
  END;

  IF op==1 THEN VER(t); END;
  IF op==2 THEN CAPTURAR(t); END;

  RETURN 1;
END;


// ---------------- MODULOS PENDIENTES ----------------

PROX(nombre)
BEGIN
  MSGBOX(nombre+" - En construccion");
  RETURN 1;
END;


// ---------------- CONFIGURACION ----------------

CONFIG()
BEGIN
  LOCAL a,d,m;

  a:=L9(7);
  d:=L9(8);
  m:=L9(9);

  IF INPUT({a,d,m},
    "Configuracion",
    {"alfa =","decimales =","modo ="},
    {"Nivel de significancia","Decimales a mostrar","1 = examen, 2 = estudio"})==0 THEN
    RETURN 0;
  END;

  IF a<=0 OR a>=1 THEN
    MSGBOX("Alfa debe estar entre 0 y 1.");
    RETURN 0;
  END;

  d:=IP(d);

  IF d<0 OR d>10 THEN
    MSGBOX("Decimales: utilizar un valor de 0 a 10.");
    RETURN 0;
  END;

  L9(7):=a;
  L9(8):=d;

  IF m==2 THEN
    L9(9):=2;
  ELSE
    L9(9):=1;
  END;

  MSGBOX("Configuracion guardada.");

  RETURN 1;
END;
