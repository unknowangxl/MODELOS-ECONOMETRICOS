
EXPORT T1EST()
BEGIN
  LOCAL res,lns,n,m,b0,b1,sg;

  INI9();

  n:=L9(1);

  IF n<3 THEN
    MSGBOX("No hay datos suficientes. Cargue los datos primero.");
    RETURN 0;
  END;

  res:=REGS(M1,1,2,n);

  IF SIZE(res)==0 THEN
    RETURN 0;
  END;

  m:=L9(9);
  b0:=res(13);
  b1:=res(12);

  lns:={};

  lns:=AGR(lns,"n = "+STRING(res(1)));

  IF m==2 THEN
    lns:=AGR(lns,"Sum x = "+FM(res(2)));
    lns:=AGR(lns,"Sum y = "+FM(res(3)));
    lns:=AGR(lns,"Sum xy = "+FM(res(4)));
    lns:=AGR(lns,"Sum x2 = "+FM(res(5)));
    lns:=AGR(lns,"Sum y2 = "+FM(res(6)));
    lns:=AGR(lns,"xbar = "+FM(res(10)));
    lns:=AGR(lns,"ybar = "+FM(res(11)));
  END;

  lns:=AGR(lns,"Sxx = "+FM(res(7)));
  lns:=AGR(lns,"Syy = "+FM(res(8)));
  lns:=AGR(lns,"Sxy = "+FM(res(9)));

  lns:=AGR(lns,"b1 = Sxy/Sxx = "+FM(b1));
  lns:=AGR(lns,"b0 = ybar-b1*xbar = "+FM(b0));

  sg:=" + ";

  IF b1<0 THEN
    sg:=" - ";
  END;

  lns:=AGR(lns,"Y^ = "+FM(b0)+sg+FM(ABS(b1))+" X");

  lns:=AGR(lns,"SST = Syy = "+FM(res(14)));
  lns:=AGR(lns,"SSR = b1*Sxy = "+FM(res(15)));
  lns:=AGR(lns,"SSE = SST-SSR = "+FM(res(16)));

  lns:=AGR(lns,"s2 = SSE/(n-2) = "+FM(res(17)));
  lns:=AGR(lns,"s = "+FM(res(18)));

  lns:=AGR(lns,"r = "+FM(res(19)));
  lns:=AGR(lns,"R2 = SSR/SST = "+FM(res(20)));

  MOSTRAR("ESTIMAR MODELO",lns);

  RETURN 1;
END;
// ===== Auxiliares internos de la Entrega 2 =====

// Valida datos y devuelve REGS de M1 ({} si hay error)
PREP()
BEGIN
  LOCAL n;
  INI9();
  n:=L9(1);
  IF n<3 THEN
    MSGBOX("No hay datos suficientes. Cargue los datos primero.");
    RETURN {};
  END;
  RETURN REGS(M1,1,2,n);
END;

// Pide alfa (valor inicial = L9(7)); devuelve 0 si cancela o es invalido
PEDALFA()
BEGIN
  LOCAL al;
  al:=L9(7);
  IF INPUT(al,"NIVEL alfa","alfa (0 a 1) =","Significancia")==0 THEN
    RETURN 0;
  END;
  IF al<=0 OR al>=1 THEN
    MSGBOX("alfa debe estar entre 0 y 1.");
    RETURN 0;
  END;
  RETURN al;
END;

// Pide un numero; devuelve {ok,valor}
PEDNUM(tit,etq,v0)
BEGIN
  LOCAL v;
  v:=v0;
  IF INPUT(v,tit,etq,"")==0 THEN
    RETURN {0,0};
  END;
  RETURN {1,v};
END;

// Prueba t bilateral de H0: par = c. Agrega lineas a lns y la devuelve.
// nom="B0", ne="b0", tr/tn = texto de conclusion si rechaza / no rechaza
TESTT(lns,nom,ne,est,se,gl,c,al,tr,tn)
BEGIN
  LOCAL tc,tcal,pv;
  tc:=TCRIT2(gl,al);
  tcal:=(est-c)/se;
  pv:=PVT2(gl,ABS(tcal));
  lns:=AGR(lns,"H0: "+nom+" = "+FM(c));
  lns:=AGR(lns,"H1: "+nom+" <> "+FM(c));
  lns:=AGR(lns,"alfa = "+FM(al));
  lns:=AGR(lns,"T=("+ne+"-valor)/EE ~ t("+STRING(gl)+")");
  lns:=AGR(lns,"Rechazar H0 si |T| > "+FM(tc));
  lns:=AGR(lns,"EE("+ne+") = "+FM(se));
  lns:=AGR(lns,"T calc = "+FM(tcal));
  lns:=AGR(lns,"p-valor = "+FM(pv));
  IF ABS(tcal)>tc THEN
    lns:=AGR(lns,"Rechazar H0");
    lns:=AGR(lns,tr);
  ELSE
    lns:=AGR(lns,"No rechazar H0");
    lns:=AGR(lns,tn);
  END;
  RETURN lns;
END;

// Inferencia sobre un parametro: IC o prueba de hipotesis
// pasos = lista de lineas extra (solo modo estudio)
PARAM(nom,ne,est,se,gl,c0,pasos)
BEGIN
  LOCAL op,al,tc,lns,k,rc,li,ls;
  LOCAL c,tr,tn;
    op:=1;
  IF CHOOSE(op,"INFERENCIA "+nom,
            "Intervalo de confianza",
            "Prueba de hipotesis")==0 THEN
    RETURN 0;
  END;
  al:=PEDALFA();
  IF al==0 THEN RETURN 0; END;
  lns:={};

  IF op==1 THEN
    tc:=TCRIT2(gl,al);
    li:=est-tc*se;
    ls:=est+tc*se;
    IF L9(9)==2 THEN
      FOR k FROM 1 TO SIZE(pasos) DO
        lns:=AGR(lns,pasos(k));
      END;
    END;
    lns:=AGR(lns,"1-alfa = "+FM(100*(1-al))+"%");
    lns:=AGR(lns,"gl = n-2 = "+STRING(gl));
    lns:=AGR(lns,"t0 = "+FM(tc));
    lns:=AGR(lns,ne+" = "+FM(est));
    lns:=AGR(lns,"EE("+ne+") = "+FM(se));
    lns:=AGR(lns,"Margen = t0*EE = "+FM(tc*se));
    lns:=AGR(lns,"LI = "+FM(li));
    lns:=AGR(lns,"LS = "+FM(ls));
    lns:=AGR(lns,"P("+FM(li)+"<="+nom+"<="+FM(ls)+")="+FM(1-al));
    MOSTRAR("IC PARA "+nom,lns);
  ELSE
    rc:=PEDNUM("VALOR DE H0","valor de "+nom+" =",c0);
    IF rc(1)==0 THEN RETURN 0; END;
    c:=rc(2);
    tr:=nom+" <> "+FM(c);
    tn:="No se descarta "+nom+" = "+FM(c);
    IF nom=="B0" AND c==0 THEN
      tr:="El modelo NO pasa por el origen";
      tn:="El modelo pasa por el origen";
    END;
    IF L9(9)==2 THEN
      FOR k FROM 1 TO SIZE(pasos) DO
        lns:=AGR(lns,pasos(k));
      END;
    END;
    lns:=TESTT(lns,nom,ne,est,se,gl,c,al,tr,tn);
    MOSTRAR("PRUEBA SOBRE "+nom,lns);
  END;
  RETURN 1;
END;

// ===== 1.1 IC / prueba para B0 =====
// EE(b0) = s*sqrt(SumX2/(n*Sxx))
EXPORT T11IC0()
BEGIN
  LOCAL res,n,se,pasos;
  res:=PREP();
  IF SIZE(res)==0 THEN RETURN 0; END;
  n:=res(1);
  se:=res(18)*SQRT(res(5)/(n*res(7)));
  pasos:={"s = "+FM(res(18)),
          "Sum x2 = "+FM(res(5)),
          "n*Sxx = "+FM(n*res(7)),
          "EE=s*sqrt(Sx2/(n*Sxx))"};
  RETURN PARAM("B0","b0",res(13),se,n-2,0,pasos);
END;

// ===== 1.2 IC / prueba para B1 =====
// EE(b1) = s/sqrt(Sxx)
EXPORT T12IC1()
BEGIN
  LOCAL res,n,se,pasos;
  res:=PREP();
  IF SIZE(res)==0 THEN RETURN 0; END;
  n:=res(1);
  se:=res(18)/SQRT(res(7));
  pasos:={"s = "+FM(res(18)),
          "Sxx = "+FM(res(7)),
          "EE=s/sqrt(Sxx)"};
  RETURN PARAM("B1","b1",res(12),se,n-2,1,pasos);
END;

// ===== 1.3 Prediccion =====
// Interpolacion: Yh +- t0*s*sqrt(1/n+(X0-xbar)^2/Sxx)
// Extrapolacion: Yh +- t0*s*sqrt(1+1/n+(X0-xbar)^2/Sxx)
EXPORT T13PRED()
BEGIN
  LOCAL res,n,al,tc,k,xmin,xmax;
  LOCAL                 rc,x0,yh,dd,se,inter,lns;
  res:=PREP();
  IF SIZE(res)==0 THEN RETURN 0; END;
  n:=res(1);
  al:=PEDALFA();
  IF al==0 THEN RETURN 0; END;
  tc:=TCRIT2(n-2,al);

  xmin:=M1(1,2);
  xmax:=M1(1,2);
  FOR k FROM 2 TO n DO
    IF M1(k,2)<xmin THEN xmin:=M1(k,2); END;
    IF M1(k,2)>xmax THEN xmax:=M1(k,2); END;
  END;

  REPEAT
    rc:=PEDNUM("PREDICCION","X0 (cancelar=salir) =",res(10));
    IF rc(1)==1 THEN
      x0:=rc(2);
      yh:=res(13)+res(12)*x0;
      dd:=(x0-res(10))^2/res(7);
      inter:=(x0>=xmin AND x0<=xmax);
      IF inter THEN
        se:=res(18)*SQRT(1/n+dd);
      ELSE
        se:=res(18)*SQRT(1+1/n+dd);
      END;
      lns:={};
      lns:=AGR(lns,"X0 = "+FM(x0));
      lns:=AGR(lns,"Region: ["+FM(xmin)+" , "+FM(xmax)+"]");
      IF inter THEN
        lns:=AGR(lns,"INTERPOLACION");
      ELSE
        lns:=AGR(lns,"EXTRAPOLACION (X0 fuera)");
      END;
      lns:=AGR(lns,"Y0^ = b0+b1*X0 = "+FM(yh));
      IF L9(9)==2 THEN
        lns:=AGR(lns,"xbar = "+FM(res(10)));
        lns:=AGR(lns,"Sxx = "+FM(res(7)));
        lns:=AGR(lns,"s = "+FM(res(18)));
        lns:=AGR(lns,"(X0-xbar)^2/Sxx = "+FM(dd));
      END;
      lns:=AGR(lns,"1-alfa = "+FM(100*(1-al))+"%");
      lns:=AGR(lns,"t0 = "+FM(tc)+" (gl="+STRING(n-2)+")");
      lns:=AGR(lns,"Margen = "+FM(tc*se));
      lns:=AGR(lns,"LI = "+FM(yh-tc*se));
      lns:=AGR(lns,"LS = "+FM(yh+tc*se));
      MOSTRAR("PREDICCION",lns);
    END;
  UNTIL rc(1)==0;
  RETURN 1;
END;

// ===== 1.4 Utilidad del modelo: prueba t de B1 = 0 =====
// T = b1/(s/sqrt(Sxx)) ~ t(n-2)   (T^2 = F de 1.5)

    RETURN 0;
  END;

  n:=res(1);

  al:=PEDALFA();

  IF al==0 THEN

  END;

    "No se demuestra poder predictivo");

  RETURN 1;
END;
// ===== 1.5 Prueba F y ANOVA =====
// F = (SSR/1)/(SSE/(n-2)) ~ F(1;n-2)
EXPORT T15F()
BEGIN
  LOCAL res,n,gl,al,cmr,cme;
  LOCAL f,f0,pv,lns;
  res:=PREP();
  IF SIZE(res)==0 THEN RETURN 0; END;
  n:=res(1);
  gl:=n-2;
  IF res(16)<=0 THEN
    MSGBOX("SSE = 0: ajuste perfecto, F no definido.");
    RETURN 0;
  END;
  al:=PEDALFA();
  IF al==0 THEN RETURN 0; END;
  cmr:=res(15);
  cme:=res(16)/gl;
  f:=cmr/cme;
  f0:=FCRIT(1,gl,al);
  pv:=PVF(1,gl,f);
  lns:={};
  lns:=AGR(lns,"H0: modelo sin poder predictivo");
  lns:=AGR(lns,"H1: modelo con poder predictivo");
  lns:=AGR(lns,"(B1=0 contra B1<>0)");
  lns:=AGR(lns,"alfa = "+FM(al));
  lns:=AGR(lns,"F = CMR/CME ~ F(1;"+STRING(gl)+")");
  lns:=AGR(lns,"Rechazar H0 si F > "+FM(f0));
  IF L9(9)==2 THEN
    lns:=AGR(lns,"SST = Syy = "+FM(res(14)));
    lns:=AGR(lns,"SSR = b1*Sxy = "+FM(res(15)));
    lns:=AGR(lns,"SSE = SST-SSR = "+FM(res(16)));
  END;
  lns:=AGR(lns,"ANOVA  SC / gl / CM");
  lns:=AGR(lns,"Regr: "+FM(res(15))+" / 1 / "+FM(cmr));
  lns:=AGR(lns,"Error: "+FM(res(16))+" / "+STRING(gl)+" / "+FM(cme));
  lns:=AGR(lns,"Total: "+FM(res(14))+" / "+STRING(n-1));
  lns:=AGR(lns,"F calc = "+FM(f));
  lns:=AGR(lns,"p-valor = "+FM(pv));
  IF f>f0 THEN
    lns:=AGR(lns,"Rechazar H0");
    lns:=AGR(lns,"El modelo SI tiene poder predictivo");
  ELSE
    lns:=AGR(lns,"No rechazar H0");
    lns:=AGR(lns,"El modelo NO tiene poder predictivo");
  END;
  MOSTRAR("PRUEBA F / ANOVA",lns);
  RETURN 1;
END;

// ===== 1.6 Prueba de linealidad (X repetidas) =====
// SSEpuro = SumY2 - Sum(Ti^2/ni);  SSEfa = SSE - SSEpuro
// F = (SSEfa/(k-2))/(SSEpuro/(n-k)) ~ F(k-2;n-k)
EXPORT T16LIN()
BEGIN
  LOCAL res,n,k,j,m,found,xs;
  LOCAL ni,ti,xv,yv,al,sq,spuro;
  LOCAL sfa,cfa,cpu,f,f0,pv,lns;
  res:=PREP();
  IF SIZE(res)==0 THEN RETURN 0; END;
  n:=res(1);

  // agrupar por valores distintos de X
  xs:={}; ni:={}; ti:={};
  FOR j FROM 1 TO n DO
    xv:=M1(j,2);
    yv:=M1(j,1);
    found:=0;
    FOR m FROM 1 TO SIZE(xs) DO
      IF xs(m)==xv THEN
        ni(m):=ni(m)+1;
        ti(m):=ti(m)+yv;
        found:=m;
      END;
    END;
    IF found==0 THEN
      xs:=CONCAT(xs,{xv});
      ni:=CONCAT(ni,{1});
      ti:=CONCAT(ti,{yv});
    END;
  END;
  k:=SIZE(xs);

  IF k==n THEN
    MSGBOX("No hay valores de X repetidos: la prueba de linealidad no se puede hacer.");
    RETURN 0;
  END;
  IF k<3 THEN
    MSGBOX("Se necesitan al menos 3 valores distintos de X (k-2>=1).");
    RETURN 0;
  END;

  sq:=0;
  FOR m FROM 1 TO k DO
    sq:=sq+ti(m)^2/ni(m);
  END;
  spuro:=res(6)-sq;
  IF spuro<=0 THEN
    MSGBOX("SSE error puro = 0: F no definido.");
    RETURN 0;
  END;
  sfa:=res(16)-spuro;
  cfa:=sfa/(k-2);
  cpu:=spuro/(n-k);
  f:=cfa/cpu;

  al:=PEDALFA();
  IF al==0 THEN RETURN 0; END;
  f0:=FCRIT(k-2,n-k,al);
  pv:=PVF(k-2,n-k,f);

  lns:={};
  lns:=AGR(lns,"H0: dependencia real lineal");
  lns:=AGR(lns,"H1: dependencia real no lineal");
  lns:=AGR(lns,"alfa = "+FM(al));
  lns:=AGR(lns,"n = "+STRING(n)+"   k = "+STRING(k));
  lns:=AGR(lns,"F ~ F("+STRING(k-2)+";"+STRING(n-k)+")");
  lns:=AGR(lns,"Rechazar H0 si F > "+FM(f0));
  IF L9(9)==2 THEN
    FOR m FROM 1 TO k DO
      lns:=AGR(lns,"X="+FM(xs(m))+" n="+STRING(ni(m))+" T="+FM(ti(m)));
    END;
    lns:=AGR(lns,"Sum Y2 = "+FM(res(6)));
    lns:=AGR(lns,"Sum Ti2/ni = "+FM(sq));
  END;
  lns:=AGR(lns,"ANOVA  SC / gl / CM");
  lns:=AGR(lns,"Regr: "+FM(res(15))+" / 1");
  lns:=AGR(lns,"Error: "+FM(res(16))+" / "+STRING(n-2));
  lns:=AGR(lns,"F.ajuste: "+FM(sfa)+" / "+STRING(k-2)+" / "+FM(cfa));
  lns:=AGR(lns,"E.puro: "+FM(spuro)+" / "+STRING(n-k)+" / "+FM(cpu));
  lns:=AGR(lns,"Total: "+FM(res(14))+" / "+STRING(n-1));
  lns:=AGR(lns,"F calc = "+FM(f));
  lns:=AGR(lns,"p-valor = "+FM(pv));
  IF f>f0 THEN
    lns:=AGR(lns,"Rechazar H0");
    lns:=AGR(lns,"La dependencia NO es lineal");
  ELSE
    lns:=AGR(lns,"No rechazar H0");
    lns:=AGR(lns,"Dependencia aprox. lineal");
  END;
  MOSTRAR("PRUEBA DE LINEALIDAD",lns);
  RETURN 1;
END;

// {min,max} de la columna c de la matriz m (n filas)
MINMAX(m,c,n)
BEGIN
  LOCAL k,mn,mx;
  mn:=m(1,c);
  mx:=m(1,c);
  FOR k FROM 2 TO n DO
    IF m(k,c)<mn THEN mn:=m(k,c); END;
    IF m(k,c)>mx THEN mx:=m(k,c); END;
  END;
  RETURN {mn,mx};
END;

// 1 si se puede linealizar (tipo 1 exponencial: Y>0; tipo 2 potencial: Y>0 y X>0)
VALIDO(tipo,n)
BEGIN
  LOCAL k;
  FOR k FROM 1 TO n DO
    IF M1(k,1)<=0 THEN RETURN 0; END;
    IF tipo==2 AND M1(k,2)<=0 THEN RETURN 0; END;
  END;
  RETURN 1;
END;

// Matriz transformada: col1 = U = ln Y ; col2 = X (tipo 1) o V = ln X (tipo 2)
// Llamar solo si VALIDO(tipo,n)==1
TRANS(tipo,n)
BEGIN
  LOCAL ma,k;
  ma:=MAKEMAT(0,n,2);
  FOR k FROM 1 TO n DO
    ma(k,1):=LN(M1(k,1));
    IF tipo==1 THEN
      ma(k,2):=M1(k,2);
    ELSE
      ma(k,2):=LN(M1(k,2));
    END;
  END;
  RETURN ma;
END;

// Estimacion del modelo no lineal. Devuelve {} si no aplica, o
// {b0,b1,R2 transformado,r transformado,R2 escala original,ln b0}
NLINFO(tipo,n,sst)
BEGIN
  LOCAL ma,rr,k,sse,yh;
  IF VALIDO(tipo,n)==0 THEN RETURN {}; END;
  ma:=TRANS(tipo,n);
  rr:=REGS(ma,1,2,n);
  IF SIZE(rr)==0 THEN RETURN {}; END;
  sse:=0;
  FOR k FROM 1 TO n DO
    yh:=EXP(rr(13)+rr(12)*ma(k,2));
    sse:=sse+(M1(k,1)-yh)^2;
  END;
  RETURN {EXP(rr(13)),rr(12),rr(20),rr(19),1-sse/sst,rr(13)};
END;

// Prediccion sobre el modelo (linealizado o no). z0 = valor de la variable
// del modelo lineal. Devuelve {U0,t0,margen,inter(1/0),(z0-zbar)^2/Szz}
PINT(res,n,z0,zmin,zmax,al)
BEGIN
  LOCAL tc,u0,dd,se,inter;
  tc:=TCRIT2(n-2,al);
  u0:=res(13)+res(12)*z0;
  dd:=(z0-res(10))^2/res(7);
  inter:=0;
  IF z0>=zmin AND z0<=zmax THEN inter:=1; END;
  IF inter==1 THEN
    se:=res(18)*SQRT(1/n+dd);
  ELSE
    se:=res(18)*SQRT(1+1/n+dd);
  END;
  RETURN {u0,tc,tc*se,inter,dd};
END;

// Comparar lineal, exponencial y potencial
COMPARA(rl,n)
BEGIN
  LOCAL lns,ie,ip,sst,r2l;
  LOCAL sg,b1,b2,v1,v2;
  sst:=rl(8);
  r2l:=rl(20);
  sg:=" + ";
  IF rl(12)<0 THEN sg:=" - "; END;
  lns:={};
  lns:=AGR(lns,"LINEAL");
  lns:=AGR(lns,"Y^ = "+FM(rl(13))+sg+FM(ABS(rl(12)))+" X");
  lns:=AGR(lns,"R2 = "+FM(r2l)+"  r = "+FM(rl(19)));
  b1:="Lineal";  v1:=r2l;
  b2:="Lineal";  v2:=r2l;

  ie:=NLINFO(1,n,sst);
  lns:=AGR(lns,"EXPONENCIAL Y=b0*e^(b1*X)");
  IF SIZE(ie)==0 THEN
    lns:=AGR(lns,"No aplica (requiere Y>0)");
  ELSE
    lns:=AGR(lns,"b0 = "+FM(ie(1))+"  b1 = "+FM(ie(2)));
    IF L9(9)==2 THEN
      lns:=AGR(lns,"ln b0 = "+FM(ie(6)));
    END;
    lns:=AGR(lns,"R2 transf = "+FM(ie(3))+"  r = "+FM(ie(4)));
    lns:=AGR(lns,"R2 esc.orig = "+FM(ie(5)));
    IF ie(3)>v1 THEN b1:="Exponencial"; v1:=ie(3); END;
    IF ie(5)>v2 THEN b2:="Exponencial"; v2:=ie(5); END;
  END;

  ip:=NLINFO(2,n,sst);
  lns:=AGR(lns,"POTENCIAL Y=b0*X^b1");
  IF SIZE(ip)==0 THEN
    lns:=AGR(lns,"No aplica (requiere X>0, Y>0)");
  ELSE
    lns:=AGR(lns,"b0 = "+FM(ip(1))+"  b1 = "+FM(ip(2)));
    IF L9(9)==2 THEN
      lns:=AGR(lns,"ln b0 = "+FM(ip(6)));
    END;
    lns:=AGR(lns,"R2 transf = "+FM(ip(3))+"  r = "+FM(ip(4)));
    lns:=AGR(lns,"R2 esc.orig = "+FM(ip(5)));
    IF ip(3)>v1 THEN b1:="Potencial"; v1:=ip(3); END;
    IF ip(5)>v2 THEN b2:="Potencial"; v2:=ip(5); END;
  END;

  lns:=AGR(lns,"MEJOR (R2 transf, libro):");
  lns:=AGR(lns,"  "+b1);
  lns:=AGR(lns,"MEJOR (R2 esc. original):");
  lns:=AGR(lns,"  "+b2);
  IF NOT(b1==b2) THEN
    lns:=AGR(lns,"Aviso: los criterios difieren");
  END;
  MOSTRAR("COMPARAR MODELOS",lns);
  RETURN 1;
END;

// Prediccion con modelo no lineal (tipo 1 exp, tipo 2 potencial)
PREDNL(rl,n,tipo)
BEGIN
  LOCAL ma,rr,al,rc,x0;
  LOCAL z0,mm,pr,lns,nmod,vn;
  IF VALIDO(tipo,n)==0 THEN
    IF tipo==1 THEN
      MSGBOX("No aplica: requiere Y>0.");
    ELSE
      MSGBOX("No aplica: requiere X>0 y Y>0.");
    END;
    RETURN 0;
  END;
  ma:=TRANS(tipo,n);
  rr:=REGS(ma,1,2,n);
  IF SIZE(rr)==0 THEN RETURN 0; END;
  al:=PEDALFA();
  IF al==0 THEN RETURN 0; END;
  mm:=MINMAX(ma,2,n);
  nmod:="EXPONENCIAL";
  vn:="X";
  IF tipo==2 THEN nmod:="POTENCIAL"; vn:="ln X"; END;

  REPEAT
    rc:=PEDNUM("PREDICCION "+nmod,"X0 (cancelar=salir) =",rl(10));
    IF rc(1)==1 THEN
      x0:=rc(2);
      IF tipo==2 AND x0<=0 THEN
        MSGBOX("X0 debe ser > 0 (potencial).");
      ELSE
        IF tipo==1 THEN z0:=x0; ELSE z0:=LN(x0); END;
        pr:=PINT(rr,n,z0,mm(1),mm(2),al);
        lns:={};
        lns:=AGR(lns,"Modelo "+nmod);
        lns:=AGR(lns,"X0 = "+FM(x0));
        IF tipo==2 THEN
          lns:=AGR(lns,"V0 = ln X0 = "+FM(z0));
        END;
        lns:=AGR(lns,"Region en "+vn+": ["+FM(mm(1))+","+FM(mm(2))+"]");
        IF pr(4)==1 THEN
          lns:=AGR(lns,"INTERPOLACION");
        ELSE
          lns:=AGR(lns,"EXTRAPOLACION (fuera)");
        END;
        IF L9(9)==2 THEN
          lns:=AGR(lns,"zbar = "+FM(rr(10)));
          lns:=AGR(lns,"Szz = "+FM(rr(7)));
          lns:=AGR(lns,"(z0-zbar)^2/Szz = "+FM(pr(5)));
        END;
        lns:=AGR(lns,"U0^ = "+FM(pr(1)));
        lns:=AGR(lns,"Su = "+FM(rr(18)));
        lns:=AGR(lns,"1-alfa = "+FM(100*(1-al))+"%");
        lns:=AGR(lns,"t0 = "+FM(pr(2))+" (gl="+STRING(n-2)+")");
        lns:=AGR(lns,"Margen (ln) = "+FM(pr(3)));
        lns:=AGR(lns,"ln LI = "+FM(pr(1)-pr(3)));
        lns:=AGR(lns,"ln LS = "+FM(pr(1)+pr(3)));
        lns:=AGR(lns,"Y0^ = e^U0 = "+FM(EXP(pr(1))));
        lns:=AGR(lns,"LI = "+FM(EXP(pr(1)-pr(3))));
        lns:=AGR(lns,"LS = "+FM(EXP(pr(1)+pr(3))));
        IF rr(20)<rl(20) THEN
          lns:=AGR(lns,"Aviso: el lineal tiene mayor R2");
        END;
        MOSTRAR("PREDICCION "+nmod,lns);
      END;
    END;
  UNTIL rc(1)==0;
  RETURN 1;
END;

// ===== 1.7 Modelos no lineales =====
EXPORT T17NOLIN()
BEGIN
  LOCAL op,rl,n;
  rl:=PREP();
  IF SIZE(rl)==0 THEN RETURN 0; END;
  n:=rl(1);
  op:=1;
  IF CHOOSE(op,"1.7 NO LINEALES",
            "Comparar modelos",
            "Predecir: exponencial",
            "Predecir: potencial")==0 THEN
    RETURN 0;
  END;
  IF op==1 THEN
    RETURN COMPARA(rl,n);
  END;
  RETURN PREDNL(rl,n,op-1);
END;
