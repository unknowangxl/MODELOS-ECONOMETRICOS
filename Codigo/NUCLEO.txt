EXPORT INI9()
BEGIN
  IF SIZE(L9)<9 THEN
    L9:={0,0,0,0,0,0,0.05,4,1};
  END;
  RETURN 1;
END;


EXPORT TCRIT2(gl,a)
BEGIN
  RETURN STUDENT_ICDF(gl,1-a/2);
END;


EXPORT TCRIT1(gl,a)
BEGIN
  RETURN STUDENT_ICDF(gl,1-a);
END;


EXPORT FCRIT(g1,g2,a)
BEGIN
  RETURN FISHER_ICDF(g1,g2,1-a);
END;


EXPORT ZCRIT2(a)
BEGIN
  RETURN NORMALD_ICDF(0,1,1-a/2);
END;


EXPORT PVT2(gl,t)
BEGIN
  RETURN 2*MIN(STUDENT_CDF(gl,t),1-STUDENT_CDF(gl,t));
END;


EXPORT PVF(g1,g2,f)
BEGIN
  RETURN 1-FISHER_CDF(g1,g2,f);
END;


EXPORT FM(x)
BEGIN
  INI9();
  RETURN STRING(ROUND(x,L9(8)));
END;


EXPORT AGR(lns,s)
BEGIN
  lns(SIZE(lns)+1):=s;
  RETURN lns;
END;


EXPORT MOSTRAR(tit,lns)
BEGIN
  LOCAL n,i,p,np,ini,fin,pp,txt;

  pp:=6;
  n:=SIZE(lns);
  np:=0;

  IF n>0 THEN
    np:=1;
    WHILE np*pp<n DO
      np:=np+1;
    END;
  END;

  FOR p FROM 1 TO np DO
    ini:=(p-1)*pp+1;
    fin:=MIN(n,p*pp);

    txt:=tit+" ("+STRING(p)+"/"+STRING(np)+")"+CHAR(10);

    FOR i FROM ini TO fin DO
      txt:=txt+lns(i)+CHAR(10);
    END;

    MSGBOX(txt);
  END;

  RETURN 1;
END;


EXPORT REGS(A,cy,cx,n)
BEGIN
  LOCAL i,x,y,sx,sy,sxy,sx2,sy2;
  LOCAL qxx,qyy,qxy,xb,yb,b1,b0;
  LOCAL sst,ssr,sse,s2,r,r2,d;

  IF n<3 THEN
    MSGBOX("Se necesitan al menos 3 observaciones.");
    RETURN {};
  END;

  IF cy==cx OR cy<1 OR cx<1 THEN
    MSGBOX("Columnas no validas.");
    RETURN {};
  END;

  d:=DIM(A);

  IF n>d(1) OR cy>d(2) OR cx>d(2) THEN
    MSGBOX("Dimensiones de matriz incompatibles.");
    RETURN {};
  END;

  sx:=0;
  sy:=0;
  sxy:=0;
  sx2:=0;
  sy2:=0;

  FOR i FROM 1 TO n DO
    x:=A(i,cx);
    y:=A(i,cy);

    sx:=sx+x;
    sy:=sy+y;
    sxy:=sxy+x*y;
    sx2:=sx2+x*x;
    sy2:=sy2+y*y;
  END;

  qxx:=sx2-sx*sx/n;
  qyy:=sy2-sy*sy/n;
  qxy:=sxy-sx*sy/n;

  IF qxx<=0.000000000001*MAX(1,sx2) THEN
    MSGBOX("Sxx es cero o demasiado pequeno.");
    RETURN {};
  END;

  IF qyy<=0.000000000001*MAX(1,sy2) THEN
    MSGBOX("Syy es cero o demasiado pequeno.");
    RETURN {};
  END;

  xb:=sx/n;
  yb:=sy/n;

  b1:=qxy/qxx;
  b0:=yb-b1*xb;

  sst:=qyy;
  ssr:=b1*qxy;
  sse:=sst-ssr;

  IF sse<0 AND ABS(sse)<0.000000001*MAX(1,sst) THEN
    sse:=0;
  END;

  IF sse<0 THEN
    MSGBOX("Error numerico al calcular SSE.");
    RETURN {};
  END;

  s2:=sse/(n-2);
  r:=qxy/SQRT(qxx*qyy);
  r2:=ssr/sst;

  RETURN {n,sx,sy,sxy,sx2,sy2,qxx,qyy,qxy,xb,yb,b1,b0,sst,ssr,sse,s2,SQRT(s2),r,r2};
END;


EXPORT CARGAEJ(k)
BEGIN
  LOCAL vx,vy,i,n;

  INI9();

  vx:={};
  vy:={};

  IF k==0 THEN
    vx:={80,100,120,140,160,180,200,220,240,260};
    vy:={70,65,90,95,110,115,120,140,155,150};
  END;

  IF k==1 THEN
    vx:={1,1.1,1.2,1.3,1.4,1.5,1.6,1.7,1.8,1.9,2};
    vy:={8.1,7.8,8.5,9.8,9.5,8.9,8.6,10.2,9.3,9.2,10.5};
  END;

  IF k==4 THEN
    vx:={2,2,4,4,6,6,8,9,9,11,11,11};
    vy:={0.8,2,2.6,3,3,4,8.1,6,7,9,10.2,10.9};
  END;

  IF k==5 THEN
    vx:={10,10,20,20,20,30,40,40,50,60,60,70,70,70,80};
    vy:={75,70,45,50,40,35,20,25,20,15,10,15,15,20,15};
  END;

  n:=SIZE(vx);

  IF n==0 THEN
    MSGBOX("Ejemplos disponibles: 0, 1, 4 o 5.");
    RETURN 0;
  END;

  M1:=MAKEMAT(0,n,2);

  FOR i FROM 1 TO n DO
    M1(i,1):=vy(i);
    M1(i,2):=vx(i);
  END;

  L9(1):=n;
  L9(2):=1;

  MSGBOX("Ejercicio "+STRING(k)+" cargado. n="+STRING(n));

  RETURN 1;
END;
